# Run from the folder containing these files.
# PowerShell:
#   .\run_benchmarks_windows.ps1

Remove-Item results.csv, results_with_speedup.csv, *.png -ErrorAction SilentlyContinue

g++ -O2 -std=c++17 -o seq.exe 1_sequential_benchmark.cpp
g++ -O2 -std=c++17 -pthread -o thr.exe 2_threaded_benchmark.cpp
g++ -O2 -std=c++17 -fopenmp -o omp.exe 3_openmp_benchmark.cpp

.\seq.exe
.\thr.exe
.\omp.exe

$nvcc = Get-Command nvcc -ErrorAction SilentlyContinue
if ($nvcc) {
    nvcc -O2 -std=c++17 -o cuda_sort.exe 4_cuda_benchmark.cu
    .\cuda_sort.exe
} else {
    Write-Host "nvcc not found; skipped CUDA benchmark. Run CUDA file on a CUDA-enabled machine."
}

python plot_results.py
